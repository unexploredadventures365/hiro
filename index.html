-- Hiro Agency Production Schema with Row Level Security (RLS)
CREATE TABLE clients (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  industry TEXT,
  fee_pct NUMERIC DEFAULT 8.33,
  guarantee_days INT DEFAULT 90,
  credit_terms TEXT DEFAULT 'Net 30'
);

CREATE TABLE jobs (
  id TEXT PRIMARY KEY,
  client_id TEXT REFERENCES clients(id),
  title TEXT NOT NULL,
  recruiter_id TEXT NOT NULL,
  min_salary NUMERIC,
  max_salary NUMERIC,
  min_exp NUMERIC,
  max_exp NUMERIC,
  skills TEXT[]
);

CREATE TABLE candidates (
  id TEXT PRIMARY KEY,
  job_id TEXT REFERENCES jobs(id),
  name TEXT NOT NULL,
  email TEXT NOT NULL,
  phone TEXT,
  ats_score NUMERIC,
  experience NUMERIC,
  current_ctc NUMERIC,
  expected_ctc NUMERIC,
  stage TEXT DEFAULT 'Sourced',
  skills TEXT[]
);

-- Recruiter Isolation RLS Policy
ALTER TABLE candidates ENABLE ROW LEVEL SECURITY;
CREATE POLICY recruiter_isolated_candidates ON candidates
FOR ALL USING (
  EXISTS (
    SELECT 1 FROM jobs 
    WHERE jobs.id = candidates.job_id 
    AND (jobs.recruiter_id = auth.uid() OR auth.jwt() ->> 'role' = 'admin')
  )
);
